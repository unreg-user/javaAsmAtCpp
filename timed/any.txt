        constexpr void set(const size_t index, Node& newNode) noexcept {
            assert(index < cache.size());

            auto* oldNode = cache[index];

            newNode.prevNode = oldNode->prevNode;
            newNode.nextNode = oldNode->nextNode;
            oldNode->prevNode->nextNode = &newNode;
            oldNode->nextNode->prevNode = &newNode;

            if constexpr (destructOldValueMode == DestructMode::DESTRUCT) {
                delete oldNode;
            } else if constexpr (destructOldValueMode == DestructMode::CLEAR) {
                oldNode->prevNode = nullptr;
                oldNode->nextNode = nullptr;
            }

            cache[index] = newNode;
