    struct dif_stack_funcs {
        using sign_trait = dstd::call::CallableTraits<FuncSign>;
        using sign_s_trait = sign_trait::ArgsTraitsT::Prepend<sign_trait::RetT>;

        using stack_f = stack_func<FuncSign>;
        using heap_f = heap_func<FuncSign>;
        using ncap_f = FuncSign*;

        using f_types = dstd::args::TypesTraits<stack_f, heap_f, ncap_f>;

        struct var : f_types::Wrap<std::variant>, sign_s_trait::Wrap<var_call> {
        };

        template <typename Callable>
        var to_var_stack(Callable& callable) {
            if constexpr (dstd::call::is_ncap<Callable>>) {
                return var{static_cast<ncap_f>(ret)};
            } else {
                return var{stack_f{callable}};
            }
        }

        template <typename Callable>
        var to_var_heap(Callable&& callable) {
            if constexpr (dstd::call::is_ncap<std::remove_cvref_t<Callable>>) {
                return var{static_cast<ncap_f>(ret)};
            } else {
                return var{heap_f{callable}};
            }
        }

            } else {
                return var{stack_f{sign_trait::ArgsTraitsT::apply_l_args([&callables] <template Args...> (Args... args) {
                    if constexpr (is_void) {
                        (callables(cstd::forward_cpy<Args>(captured_args)...), ...);

                    }
                })}};
            }
        }

                    } else {
                        return std::tuple{ Callables{}(cstd::forward_cpy<Args>(captured_args)...)... };

                })}};

    template <typename ElType, typename Sym, size_t span_size>
    struct span_to_syms_iterator {
        static_assert(sizeof(ElType) % sizeof(Sym) == 0, "Invalid types size alignment");

        static constexpr size_t SymsPerEl = sizeof(ElType) / sizeof(Sym);
        using ElTypeAsSymArr = Sym[SymsPerEl];

        using iterator_category = std::random_access_iterator_tag;
        using iterator_concept = std::random_access_iterator_tag;
        using value_type = Sym;
        using difference_type = std::ptrdiff_t;
        using pointer = Sym*;
        using reference = Sym&;

        std::span<ElType, span_size> obj;
        size_t i;

        NODISCARD decltype(auto) operator*(this auto&& self) noexcept {
            size_t el_idx = self.i / SymsPerEl;
            size_t sym_idx = self.i % SymsPerEl;

            auto* arr_ptr = reinterpret_cast<ElTypeAsSymArr*>(&self.obj[el_idx]);
            return (*arr_ptr)[sym_idx];
        }

        NODISCARD decltype(auto) operator->(this auto&& self) noexcept {
            return &*self;
        }

        span_to_syms_iterator& operator++() noexcept {
            i++;
            return *this;
        }

        span_to_syms_iterator operator++(int) noexcept {
            span_to_syms_iterator old = *this;
            i++;
            return old;
        }

        span_to_syms_iterator& operator--() noexcept {
            i--;
            return *this;
        }

        span_to_syms_iterator operator--(int) noexcept {
            span_to_syms_iterator old = *this;
            i--;
            return old;
        }

        span_to_syms_iterator& operator+=(const difference_type n) noexcept {
            i += n;
            return *this;
        }

        span_to_syms_iterator& operator-=(const difference_type n) noexcept {
            i -= n;
            return *this;
        }

        NODISCARD friend span_to_syms_iterator operator+(span_to_syms_iterator it, difference_type n) noexcept {
            it += n;
            return it;
        }

        NODISCARD friend span_to_syms_iterator operator+(difference_type n, span_to_syms_iterator it) noexcept {
            it += n;
            return it;
        }

        NODISCARD friend span_to_syms_iterator operator-(span_to_syms_iterator it, difference_type n) noexcept {
            it -= n;
            return it;
        }

        NODISCARD friend difference_type operator-(const span_to_syms_iterator& a,
                                                   const span_to_syms_iterator& b) noexcept {
            return static_cast<difference_type>(a.i) - static_cast<difference_type>(b.i);
        }

        NODISCARD decltype(auto) operator[](this auto&& self, difference_type n) const noexcept {
            return *(self + n);
        }

        NODISCARD auto operator<=>(const span_to_syms_iterator& other) const noexcept {
            return i <=> other.i;
        }

        NODISCARD bool operator==(const span_to_syms_iterator& other) const noexcept {
            return i == other.i;
        }
    };

    template <typename Sym, typename ElType, size_t span_size>
    struct iterate_span_by_syms {
        static_assert(sizeof(ElType) % sizeof(Sym) == 0, "Invalid types size alignment");
        std::span<ElType, span_size> t;

        static constexpr size_t SymsPerEl = sizeof(ElType) / sizeof(Sym);

        decltype(auto) begin(this auto&& self) noexcept {
            return span_to_syms_iterator<ElType, Sym, span_size>{self.t, 0};
        }

        decltype(auto) end(this auto&& self) noexcept {
            return span_to_syms_iterator<ElType, Sym, span_size>{self.t, SymsPerEl};
        }
    };
